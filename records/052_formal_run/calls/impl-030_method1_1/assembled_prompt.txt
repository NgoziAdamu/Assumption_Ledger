#!/usr/env/bin python3
import os
import yaml
import duckdb
from pathlib import Path
from typing import List, Optional
import logging

logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')
logger = logging.getLogger(__name__)

class DataPipeline:
    def __init__(self,
                 source_db_path: str = "zuora_start.duckdb",
                 target_db_path: str = "zuora.duckdb",
                 config_path: str = "config/layer_dependencies.yaml"):
        self.source_db_path = source_db_path
        self.target_db_path = target_db_path
        self.config_path = config_path
        self.sql_dir = Path("sql")
        self.schema_config = {
            'raw': 'raw',
            'staging': 'staging',
            'intermediate': 'intermediate',
            'marts': 'marts',
        }
        self.conn = None
        self.config = self._load_config()

    def _load_config(self) -> dict:
        try:
            with open(self.config_path, 'r', encoding='utf-8') as f:
                return yaml.safe_load(f)
        except Exception as e:
            logger.error(f"Failed to load configuration: {e}")
            raise

    def _create_schemas(self):
        for _, schema_name in self.schema_config.items():
            self.conn.execute(f"CREATE SCHEMA IF NOT EXISTS {schema_name}")

    def _import_raw_data(self):
        logger.info("Importing raw schema from source database (priority)")
  
        try:
            self.conn.execute(f"ATTACH '{self.source_db_path}' AS src_db")
            rows = self.conn.execute(
                "SELECT table_name FROM information_schema.tables "
                "WHERE table_catalog='src_db' AND table_schema='raw' AND table_type='BASE TABLE'"
            ).fetchall()
            tables = [r[0] for r in rows]
            if not tables:
                raise Exception("No tables found in the source database's raw schema")
            for tbl in tables:
          
                sinfo = self.conn.execute(f"PRAGMA table_info('src_db.raw.{tbl}')").fetchall()
                if len(sinfo) == 1:
                    header = sinfo[0][1]                                                           
                    header_cols = [c.strip() for c in header.split(',') if c.strip()]
              
                    col_ident = '"' + header.replace('"','""') + '"'
                    split_expr = f"string_split({col_ident}, ',')"
                    select_cols = ",\n".join([
                        f"trim(list_extract({split_expr}, {i})) AS {col}"
                        for i, col in enumerate(header_cols, start=1)
                    ])
                    ddl = f"CREATE OR REPLACE TABLE raw.{tbl} AS SELECT {select_cols} FROM src_db.raw.{tbl}"
                   
                    self.conn.execute(ddl)
                else:
                    self.conn.execute(f"CREATE OR REPLACE TABLE raw.{tbl} AS SELECT * FROM src_db.raw.{tbl}")
                cnt = self.conn.execute(f"SELECT COUNT(*) FROM raw.{tbl}").fetchone()[0]
                logger.info(f"  raw.{tbl}: {cnt} rows (from src_db)")
        except Exception as e:
            logger.error(f"Failed to import from source database: {e}")
            raise
        finally:
            try:
                self.conn.execute("DETACH src_db")
            except Exception:
                pass

    def _load_sql(self, path: Path) -> str:
        with open(path, 'r', encoding='utf-8') as f:
            return f.read()

    def _execute_sql(self, sql: str, table_name: str, layer: str) -> bool:
        try:
            target = f"{self.schema_config[layer]}.{table_name}"
            create_sql = f"CREATE OR REPLACE TABLE {target} AS ({sql})"
            logger.info(f"Executing {target}")
            self.conn.execute(create_sql)
            cnt = self.conn.execute(f"SELECT COUNT(*) FROM {target}").fetchone()[0]
            logger.info(f"  ✓ {target}: {cnt} rows")
            return True
        except Exception as e:
            logger.error(f"  ✗ {table_name}: {e}")
            return False

    def _execute_layer(self, layer: str) -> bool:
        layer_cfg = self.config['layers'][layer]
        tables = layer_cfg.get('tables', [])
        order = self.config.get('execution_order', {}).get(layer, tables)
        ok = True
        for t in order:
            sql_path = self.sql_dir / layer / f"{t}.sql"
            if not sql_path.exists():
                logger.warning(f"Skipping {t}: missing {sql_path}")
                ok = False
                continue
            sql = self._load_sql(sql_path)
            ok = self._execute_sql(sql, t, layer) and ok
        return ok

    def _clean_generated(self):
        for s in ['staging','intermediate','marts']:
            try:
                self.conn.execute(f"DROP SCHEMA IF EXISTS {s} CASCADE")
            except Exception as e:
                logger.warning(f"Failed to delete {s}: {e}")

    def run(self, layers: Optional[List[str]] = None) -> bool:
        if layers is None:
            layers = list(self.config['layers'].keys())
        logger.info(f"Source DB: {self.source_db_path}  Target DB: {self.target_db_path}")
        self.conn = duckdb.connect(self.target_db_path)
        self._clean_generated()
        self._create_schemas()
  
        self._import_raw_data()
        total = True
        for layer in layers:
            deps = self.config['layers'][layer].get('dependencies', [])
            for d in deps:
                if d not in layers[:layers.index(layer)]:
                    logger.warning(f"{layer} depends on {d}, but it was not executed first")
            ok = self._execute_layer(layer)
            total = total and ok
        return total


def main():
    pipeline = DataPipeline()
    success = pipeline.run()
    if success:
        print("\n✅ Zuora data warehouse built successfully")
        print(f"Source: zuora_start.duckdb -> Target: zuora.duckdb")
    else:
        print("\n❌ Execution failed, please check the logs")

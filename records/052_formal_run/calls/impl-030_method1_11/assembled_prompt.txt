## int_zuora__transactions_grouped

    - name: int_zuora__transactions_grouped
      description: Daily Zuora invoice-based transaction aggregates per account with day/week/month/year buckets.
      grain: One row per account_id per date_day
      source_models:
      - staging.stg_zuora__invoice
      - staging.stg_zuora__invoice_item
      business_logic: '- Keep only records flagged as most recent from both invoices and invoice items. - Invoices use: invoice_id, account_id, invoice_date, amount, payment_amount, refund_amount. - Invoice items use: invoice_id, invoice_item_id. - For each invoice, count unique invoice_item_id values to get its item count; null item IDs do not contribute. - Left join per-invoice item counts to invoices by invoice_id so invoices without items are retained. - Derive calendar buckets from invoice_date: - date_day: the calendar date (time removed). - date_week: Monday of the week containing the date. - date_month: first day of the calendar month. - date_year: first day of the calendar year. - Group by account_id and the four time buckets (equivalent to account_id by date_day) and compute: - daily_invoices: number of unique invoices in the group; missing invoice IDs do not count. - daily_invoice_items: sum of each invoice’s unique item count; invoices with no items contribute 0; if none have items the result is 0. - daily_invoice_amount: total of invoice amounts; null amounts are ignored; result is null if no non-null amounts exist in the group. - daily_invoice_amount_paid: total of payment amounts; nulls are ignored; result is null if all are null. - daily_invoice_amount_unpaid: for each invoice take amount minus payment amount, then total; if either value is null for an invoice it contributes nothing; result is null if all are null. - daily_refunds: total of refund amounts; nulls are ignored; result is null if all are null. - Output one row per account_id and date_day; other date buckets are deterministic functions of invoice_date for that row.'
      columns:
      - name: account_id
        data_type: VARCHAR
        description: Zuora account identifier used to group aggregates.
        tests:
        - not_null
      - name: date_day
        data_type: DATE
        description: Calendar date from invoice_date with time removed.
        tests:
        - not_null
      - name: date_week
        data_type: DATE
        description: Monday of the week containing invoice_date.
      - name: date_month
        data_type: DATE
        description: First day of the month containing invoice_date.
      - name: date_year
        data_type: DATE
        description: First day of the year containing invoice_date.
      - name: daily_invoices
        data_type: BIGINT
        description: Distinct invoice count for the account and date_day; missing invoice IDs do not count.
      - name: daily_invoice_items
        data_type: HUGEINT
        description: Sum of per-invoice unique item counts; invoices with no items contribute 0 and result is 0 if none have items.
      - name: daily_invoice_amount
        data_type: DOUBLE
        description: Total invoice amounts for the group; null amounts are ignored and result is null if all are null.
      - name: daily_invoice_amount_paid
        data_type: DOUBLE
        description: Total payment amounts for the group; nulls are ignored and result is null if all are null.
      - name: daily_invoice_amount_unpaid
        data_type: DOUBLE
        description: Total of per-invoice (amount minus payment); invoices with null in either value contribute nothing; result is null if all are null.
      - name: daily_refunds
        data_type: DOUBLE
        description: Total refund amounts for the group; nulls are ignored and result is null if all are null.

WITH joined AS (
    SELECT ii.invoice_id AS header_id, ii.invoice_item_id AS line_item_id,
           ii.created_date AS line_item_created_at, inv.created_at AS invoice_created_at,
           inv.status AS header_status, inv.purchase_type AS billing_type,
           ii.transaction_currency AS currency, ii.product_id, prod.name AS product_name,
           prod.category AS product_type,
           CASE ii.processing_type WHEN '0' THEN 'charge' WHEN '1' THEN 'discount'
                WHEN '2' THEN 'prepayment' WHEN '3' THEN 'tax' ELSE NULL END AS transaction_type,
           ii.quantity, ii.unit_price AS unit_amount,
           CASE WHEN ii.processing_type = '1' THEN try_cast(ii.charge_amount AS DOUBLE) ELSE 0.0 END AS discount_amount,
           ii.tax_amount, ii.charge_amount AS total_amount,
           ip.payment_id, ip.payment_method_id AS invoice_payment_method_id,
           p.payment_method_id AS payment_method_id, pm.payment_method_name AS payment_method,
           p.effective_date AS payment_at, inv.refund_amount,
           ii.subscription_id, rp.name AS subscription_plan,
           try_cast(s.subscription_start_date AS TIMESTAMP) AS subscription_period_started_at,
           try_cast(s.subscription_end_date AS TIMESTAMP) AS subscription_period_ended_at,
           s.status AS subscription_status, ii.account_id AS customer_id,
           'customer' AS customer_level, a.created_date AS customer_created_at,
           a.name AS customer_company, c.first_name || ' ' || c.last_name AS customer_name,
           c.work_email AS customer_email, c.city AS customer_city, c.country AS customer_country
    FROM intermediate.int_zuora__invoice_item_base AS ii
    LEFT JOIN intermediate.int_zuora__invoice_base AS inv ON ii.invoice_id = inv.invoice_id
    LEFT JOIN intermediate.int_zuora__invoice_payment_current AS ip ON ii.invoice_id = ip.invoice_id
    LEFT JOIN intermediate.int_zuora__payment_current AS p ON ip.payment_id = p.payment_id
    LEFT JOIN intermediate.int_zuora__payment_method_current AS pm ON p.payment_method_id = pm.payment_method_id
    LEFT JOIN intermediate.int_zuora__account_current AS a ON ii.account_id = a.account_id
    LEFT JOIN intermediate.int_zuora__contact_current AS c ON ii.account_id = c.account_id
    LEFT JOIN intermediate.int_zuora__product_current AS prod ON ii.product_id = prod.product_id
    LEFT JOIN intermediate.int_zuora__subscription_current AS s ON ii.subscription_id = s.subscription_id
    LEFT JOIN intermediate.int_zuora__rate_plan_current AS rp ON ii.subscription_id = rp.subscription_id
)
SELECT *, row_number() OVER (PARTITION BY header_id ORDER BY line_item_id) AS line_item_index
FROM joined

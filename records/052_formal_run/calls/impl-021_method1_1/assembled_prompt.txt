#!/usr/bin/env python3
import os
import yaml
import duckdb
from pathlib import Path
import logging
from typing import List, Optional

logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')
logger = logging.getLogger(__name__)

class DataPipeline:
    def __init__(self,
                 source_db_path: str = "klaviyo_start.duckdb",
                 target_db_path: str = "klaviyo.duckdb",
                 config_path: str = "config/layer_dependencies.yaml"):
        self.source_db_path = source_db_path
        self.target_db_path = target_db_path
        self.config_path = config_path
        self.sql_dir = Path("sql")
        self.conn = None
        self.config = self._load_config()
        self.schema_config = {
            'raw': 'raw',
            'staging': 'staging',
            'intermediate': 'intermediate',
            'marts': 'marts'
        }

    def _load_config(self) -> dict:
        with open(self.config_path, 'r', encoding='utf-8') as f:
            return yaml.safe_load(f)

    def _create_schemas(self):
        for schema in self.schema_config.values():
            self.conn.execute(f"CREATE SCHEMA IF NOT EXISTS {schema}")

    def _import_raw(self):
        logger.info("Importing raw tables from source DB")
        if not os.path.exists(self.source_db_path):
            raise FileNotFoundError(f"Source DB not found: {self.source_db_path}")
        src = duckdb.connect(self.source_db_path)
        try:
            tables = [t[0] for t in src.execute("SHOW TABLES").fetchall()]
            for t in tables:
                cols = src.execute(f"DESCRIBE {t}").fetchall()
                col_defs = []
                for name, dtype, *_ in cols:
                    safe = name if name.lower() not in ['end','start','order','group','select','from','where'] else f'"{name}"'
                    col_defs.append(f"{safe} {dtype}")
                self.conn.execute(f"CREATE OR REPLACE TABLE raw.{t} ({', '.join(col_defs)})")
                data = src.execute(f"SELECT * FROM {t}").fetchall()
                if data:
                    placeholders = ', '.join(['?' for _ in cols])
                    self.conn.executemany(f"INSERT INTO raw.{t} VALUES ({placeholders})", data)
                logger.info(f"  ✓ raw.{t}: {len(data)} rows")
        finally:
            src.close()

    def _load_sql(self, path: Path) -> str:
        with open(path, 'r', encoding='utf-8') as f:
            return f.read()

    def _execute_sql(self, sql: str, table: str, layer: str) -> bool:
        target_schema = self.schema_config[layer]
        full_name = f"{target_schema}.{table}"
        try:
            self.conn.execute(f"CREATE OR REPLACE TABLE {full_name} AS ({sql})")
            cnt = self.conn.execute(f"SELECT COUNT(*) FROM {full_name}").fetchone()[0]
            logger.info(f"✓ {full_name}: {cnt} rows")
            return True
        except Exception as e:
            logger.error(f"✗ {full_name}: {e}")
            return False

    def _execute_layer(self, layer: str) -> bool:
        layer_conf = self.config['layers'][layer]
        tables: List[str] = layer_conf.get('tables', [])
        order: List[str] = self.config.get('execution_order', {}).get(layer, tables)
        ok = True
        logger.info(f"\n=== Executing {layer.upper()} layer ===")
        for tbl in order:
            sql_path = self.sql_dir / layer / f"{tbl}.sql"
            if not sql_path.exists() and layer == 'intermediate':
          
                sql_path = self.sql_dir / layer / f"{tbl}.sql"
            if not sql_path.exists():
                logger.warning(f"Skip {tbl}: SQL not found at {sql_path}")
                ok = False
                continue
            sql = self._load_sql(sql_path)
            ok = self._execute_sql(sql, tbl, layer) and ok
        return ok

    def _clean_generated(self):
        for s in ['staging','intermediate','marts']:
            try:
                self.conn.execute(f"DROP SCHEMA IF EXISTS {s} CASCADE")
            except Exception:
                pass

    def run(self, layers: Optional[List[str]] = None) -> bool:
        if layers is None:
            layers = list(self.config['layers'].keys())
        self.conn = duckdb.connect(self.target_db_path)
        self._clean_generated()
        self._create_schemas()
  
        self._import_raw()
        success = True
        for layer in layers:
            deps = self.config['layers'][layer].get('dependencies', [])
            for d in deps:
                if d not in layers[:layers.index(layer)]:
                    logger.warning(f"{layer} depends on {d} but {d} not scheduled before it")
            success = self._execute_layer(layer) and success
        return success


def main():
    pipeline = DataPipeline()
    if pipeline.run():
        print("\n✅ Data transformation completed successfully!")
        print(f"Source: {pipeline.source_db_path} -> Target: {pipeline.target_db_path}")
    else:
        print("\n❌ Execution failed, please check the logs")

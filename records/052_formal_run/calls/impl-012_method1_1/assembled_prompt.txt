    def run_pipeline(self, layers: Optional[List[str]] = None) -> bool:
        """Run the complete data pipeline"""
        if layers is None:
            layers = self.config['execution_order']
        
        logger.info("🚀 Starting Marketo data pipeline execution")
        logger.info(f"Source database: {self.source_db_path}")
        logger.info(f"Target database: {self.target_db_path}")
        
          
        if not os.path.exists(self.target_db_path):
            logger.info("Target database does not exist, starting initialization...")
            try:
          
                self._initialize_database()
            except Exception as e:
                logger.error(f"Failed to initialize database: {e}")
                return False
        else:
            logger.info("Existing target database found, using it directly...")
      
            self.conn = duckdb.connect(self.target_db_path)
        
        self._clean_generated_tables()
        
          
        self._create_schemas()
        
        total_success = True
        
          
        execution_layers = [layer for layer in layers if layer != 'raw']
        
        for layer_name in execution_layers:
      
            layer_success = self._execute_layer(layer_name)
            total_success = total_success and layer_success
        
        if total_success:
            logger.info("\n🎉 Marketo data pipeline execution complete!")
            logger.info(f"✨ A complete data warehouse has been generated: {self.target_db_path}")
            self._show_final_stats()
        else:
            logger.error("\n❌ Errors occurred during data pipeline execution")
        
        if self.conn:
            self.conn.close()
        
        return total_success
    
    def _show_final_stats(self):
        """Display final statistics - shown by schema layer"""
        logger.info("\n=== Final Data Statistics (by Schema Layer) ===")
        
          
        for layer_name, schema_name in self.schema_config.items():
            try:
          
                tables_result = self.conn.execute(f"""
                    SELECT table_name 
                    FROM information_schema.tables 
                    WHERE table_schema = '{schema_name}'
                """).fetchall()
                schema_tables = [table[0] for table in tables_result]
                
                if schema_tables:
                    logger.info(f"\n{layer_name.upper()} Layer (Schema: {schema_name}):")
                    for table in sorted(schema_tables):

                    logger.info(f"\n{layer_name.upper()} Layer (Schema: {schema_name}): No tables")

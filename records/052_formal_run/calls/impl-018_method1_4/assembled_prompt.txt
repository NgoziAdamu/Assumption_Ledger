with lead_base as (

),
scored as (
    select *,
        trim(concat(coalesce(first_name, ''), ' ', coalesce(last_name, ''))) as derived_lead_name,
        (case when email is not null and strpos(email, '@') > 0 and strpos(email, '.') > 0 then 20 else 0 end
         + case when nullif(trim(phone), '') is not null then 15 else 0 end
         + case when nullif(trim(mobile_phone), '') is not null then 5 else 0 end
         + case when nullif(trim(industry), '') is not null then 10 else 0 end
         + case when annual_revenue > 0 then 15 else 0 end
         + case when number_of_employees > 0 then 10 else 0 end
         + case when lower(lead_source) in ('referral','website','web','partner referral','word of mouth') then 20 else 10 end
         + case when nullif(trim(company), '') is not null and lower(trim(company)) not in ('test','example','sample','demo','unknown','n/a','none') then 5 else 0 end)::integer as lead_quality_score,
        case when lower(lead_source) in ('web','website','online','social media','email','google ads','linkedin') then 'Digital'
             when lower(lead_source) in ('referral','partner referral','word of mouth','employee referral') then 'Referral'
             when lower(lead_source) in ('trade show','event','conference','purchased list','advertising') then 'Marketing'
             when lower(lead_source) in ('phone inquiry','direct mail','cold call') then 'Direct' else 'Other' end as lead_source_category,
        case when is_converted then date_diff('day', created_date::date, converted_date::date)
             when created_date is not null then date_diff('day', created_date::date, current_date) end as lead_age_days,
        case when is_converted and converted_date is not null then date_diff('day', created_date::date, converted_date::date) end as conversion_time_days,
        (coalesce(is_converted, false) and converted_account_id is not null and converted_contact_id is not null and converted_opportunity_id is not null) as has_complete_conversion
    from lead_base
),
finalized as (
    select *, case when lead_quality_score >= 80 then 'Hot' when lead_quality_score >= 60 then 'Warm'
                   when lead_quality_score >= 40 then 'Cool' else 'Cold' end as lead_temperature,
        case when conversion_time_days <= 7 then 'Fast Track' when conversion_time_days <= 30 then 'Standard'
             when conversion_time_days <= 90 then 'Slow' when conversion_time_days > 90 then 'Extended' end as conversion_speed_category
    from scored
)

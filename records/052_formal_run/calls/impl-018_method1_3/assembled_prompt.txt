    coalesce(m.activity_trend_30d, 'New') as activity_trend_30d,
    coalesce(m.is_declining_engagement, false) as is_declining_engagement,
    m.most_active_user_id, coalesce(m.activity_diversity_score, 0) as activity_diversity_score
from active_accounts a
left join activity_metrics m on a.account_id = m.account_id
left join staging.stg_salesforce__account s on a.account_id = s.account_id

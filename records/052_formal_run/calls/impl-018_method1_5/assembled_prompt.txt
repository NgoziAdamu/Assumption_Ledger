with enhanced_opportunity as (
    select o.*, u.manager_id,
        case when o.is_won then 'Won' when o.is_won = false and o.is_closed then 'Lost'
             when o.is_closed = false and lower(o.forecast_category) in ('pipeline','forecast','bestcase') then 'Pipeline' end as deal_status,
        case when o.is_created_this_month then o.amount else 0 end as created_amount_month,
        case when o.is_created_this_quarter then o.amount else 0 end as created_amount_quarter,
        case when o.is_created_this_month then 1 else 0 end as created_count_month,
        case when o.is_created_this_quarter then 1 else 0 end as created_count_quarter,
        case when o.is_closed_this_month then o.amount else 0 end as closed_amount_month,
        case when o.is_closed_this_quarter then o.amount else 0 end as closed_amount_quarter,
        case when o.is_closed_this_month then 1 else 0 end as closed_count_month,
        case when o.is_closed_this_quarter then 1 else 0 end as closed_count_quarter,
        o.amount * o.probability as weighted_amount
    from staging.stg_salesforce__opportunity o
    left join staging.stg_salesforce__user u on o.owner_id=u.user_id
    where not coalesce(o.is_deleted,false)
),
bookings as (
 select owner_id, manager_id, round(sum(closed_amount_month)) as bookings_amount_closed_this_month,
 round(sum(closed_amount_quarter)) as bookings_amount_closed_this_quarter, count(*) as total_number_bookings,
 round(sum(amount)) as total_bookings_amount, sum(closed_count_month) as bookings_count_closed_this_month,
 sum(closed_count_quarter) as bookings_count_closed_this_quarter, round(avg(amount))::double as avg_bookings_amount,
 max(amount) as largest_booking, avg(days_to_close)::double as avg_days_to_close
 from enhanced_opportunity where deal_status='Won' group by owner_id,manager_id
),
lost as (
 select owner_id, manager_id, round(sum(closed_amount_month)) as lost_amount_this_month,
 round(sum(closed_amount_quarter)) as lost_amount_this_quarter, count(*) as total_number_lost,
 round(sum(amount)) as total_lost_amount, sum(closed_count_month) as lost_count_this_month,
 sum(closed_count_quarter) as lost_count_this_quarter
 from enhanced_opportunity where deal_status='Lost' group by owner_id,manager_id
),
pipeline as (
 select owner_id, manager_id, round(sum(created_amount_month)) as pipeline_created_amount_this_month,
 round(sum(created_amount_quarter)) as pipeline_created_amount_this_quarter,
 round(sum(created_amount_month*probability)) as pipeline_created_forecast_amount_this_month,
 round(sum(created_amount_quarter*probability)) as pipeline_created_forecast_amount_this_quarter,
 sum(created_count_month) as pipeline_count_created_this_month,sum(created_count_quarter) as pipeline_count_created_this_quarter,
 count(*) as total_number_pipeline,round(sum(amount)) as total_pipeline_amount,round(sum(weighted_amount)) as total_pipeline_forecast_amount,
 round(avg(amount))::double as avg_pipeline_opp_amount,max(amount) as largest_deal_in_pipeline,avg(days_since_created)::double as avg_days_open
 from enhanced_opportunity where deal_status='Pipeline' group by owner_id,manager_id
)
select u.user_id as owner_id, coalesce(p.manager_id,b.manager_id,l.manager_id) as manager_id,
 b.manager_id as b_manager_id,b.owner_id as b_owner_id,b.bookings_amount_closed_this_month,b.bookings_amount_closed_this_quarter,
 b.total_number_bookings,b.total_bookings_amount,b.bookings_count_closed_this_month,b.bookings_count_closed_this_quarter,b.avg_bookings_amount,b.largest_booking,b.avg_days_to_close,
 l.manager_id as l_manager_id,l.owner_id as l_owner_id,l.lost_amount_this_month,l.lost_amount_this_quarter,l.total_number_lost,l.total_lost_amount,l.lost_count_this_month,l.lost_count_this_quarter,
 p.manager_id as p_manager_id,p.owner_id as p_owner_id,p.pipeline_created_amount_this_month,p.pipeline_created_amount_this_quarter,
 p.pipeline_created_forecast_amount_this_month,p.pipeline_created_forecast_amount_this_quarter,p.pipeline_count_created_this_month,p.pipeline_count_created_this_quarter,
 p.total_number_pipeline,p.total_pipeline_amount,p.total_pipeline_forecast_amount,p.avg_pipeline_opp_amount,p.largest_deal_in_pipeline,p.avg_days_open
from staging.stg_salesforce__user u
left join bookings b on u.user_id=b.owner_id left join lost l on u.user_id=l.owner_id left join pipeline p on u.user_id=p.owner_id

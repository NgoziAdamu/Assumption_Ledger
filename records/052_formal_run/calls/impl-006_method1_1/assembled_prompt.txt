#!/usr/bin/env python3
import os
import yaml
import duckdb
from pathlib import Path
from typing import List, Optional
import logging
import re


logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s'
)
logger = logging.getLogger(__name__)


class AdReportingPipeline:
    """Ad Data Pipeline Orchestrator - Supports a layered schema architecture"""
    
    def __init__(self, 
                 source_db_path: str = "ad_reporting_start.duckdb", 
                 target_db_path: str = "ad_reporting.duckdb",
                 config_path: str = "config/layer_dependencies.yaml"):
        self.source_db_path = source_db_path
        self.target_db_path = target_db_path
        self.config_path = config_path
        self.config = self._load_config()
        self.sql_dir = Path("sql")
        
                
        self.schema_config = {
            'raw': 'raw',                           
            'staging': 'staging',                   
            'intermediate': 'intermediate',             
            'marts': 'marts'                        
        }
        
                
        self.conn = None
        
    def _load_config(self) -> dict:
        """Load configuration file"""
        try:
            with open(self.config_path, 'r', encoding='utf-8') as f:
                return yaml.safe_load(f)
        except FileNotFoundError:
            logger.error(f"Configuration file {self.config_path} not found")
            raise
        except yaml.YAMLError as e:
            logger.error(f"Error parsing configuration file: {e}")
            raise
    
    def _initialize_database(self):
        """Initialize the working database, create the layered schema architecture, and import raw data"""
        logger.info(f"📥 Creating new layered database: {self.target_db_path}")
        
                
        if not os.path.exists(self.source_db_path):
            raise FileNotFoundError(f"Source database file not found: {self.source_db_path}")
        
        self.conn = duckdb.connect(self.target_db_path)
        
                
        self._create_schemas()
        
                
        self._import_raw_data()
        
        logger.info(f"✓ Layered database initialization complete: {self.target_db_path}")
    
    def _create_schemas(self):
        """Create schemas for data layers"""
        logger.info("🏗️ Creating Schema architecture layers")
        
        for layer, schema_name in self.schema_config.items():
            try:
                self.conn.execute(f"CREATE SCHEMA IF NOT EXISTS {schema_name}")
                logger.info(f"  ✓ Created schema: {schema_name}")
            except Exception as e:
                logger.error(f"  ✗ Failed to create schema {schema_name}: {e}")
                raise
    
    def _import_raw_data(self):
        """Import raw data from the source database to the raw schema"""
        logger.info("📥 Importing raw data to raw schema")
        
                
        try:
            source_conn = duckdb.connect(self.source_db_path)
        except Exception as e:
            logger.error(f"Could not connect to source database {self.source_db_path}: {e}")
            raise
        
        try:
            
            tables_result = source_conn.execute("SELECT table_name FROM information_schema.tables WHERE table_schema = 'raw'").fetchall()
            source_tables = [table[0] for table in tables_result]
            
            logger.info(f"Found {len(source_tables)} raw tables")
            
                        
            for table_name in source_tables:
                try:
                    
                    data = source_conn.execute(f"SELECT * FROM raw.{table_name}").fetchall()
                    columns_info = source_conn.execute(f"DESCRIBE raw.{table_name}").fetchall()
                    
                                        
                    column_defs = []
                    for col_info in columns_info:
                        col_name, col_type = col_info[0], col_info[1]
                        
                        if col_name.lower() in ['end', 'start', 'order', 'group', 'select', 'from', 'where']:
                            safe_col_name = f'"{col_name}"'
                        else:
                            safe_col_name = col_name
                        column_defs.append(f"{safe_col_name} {col_type}")
                    
                    create_sql = f"CREATE OR REPLACE TABLE raw.{table_name} ({', '.join(column_defs)})"
                    self.conn.execute(create_sql)
                    
                                        
                    if data:
                        placeholders = ', '.join(['?' for _ in columns_info])
                        insert_sql = f"INSERT INTO raw.{table_name} VALUES ({placeholders})"
                        self.conn.executemany(insert_sql, data)
                    
                    logger.info(f"  ✓ raw.{table_name}: {len(data)} rows")
                    
                except Exception as e:
                    logger.error(f"  ✗ Failed to import table {table_name}: {e}")
        
        finally:
            source_conn.close()
        
        logger.info("✓ Raw data import complete")
    
    def _load_sql_file(self, file_path: Path) -> str:
        """Load SQL file content"""
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                return f.read()
        except FileNotFoundError:
            logger.error(f"SQL file {file_path} not found")
            raise
    
    def _execute_sql(self, sql: str, table_name: str, layer_name: str) -> bool:
        """Execute SQL and create a table in the specified schema"""
        try:
            
            target_schema = self.schema_config[layer_name]
            full_table_name = f"{target_schema}.{table_name}"
            
                        
            processed_sql = self._process_sql_references(sql, layer_name)
            
                        
            create_table_sql = f"CREATE OR REPLACE TABLE {full_table_name} AS {processed_sql}"
            
            logger.info(f"Executing: {full_table_name}")
            self.conn.execute(create_table_sql)
            
                        
            count_result = self.conn.execute(f"SELECT COUNT(*) FROM {full_table_name}").fetchone()
            row_count = count_result[0] if count_result else 0
            
            logger.info(f"✓ {full_table_name}: {row_count:,} rows")
            return True
            
        except Exception as e:
            logger.error(f"✗ {full_table_name}: {e}")
            return False
    
    def _process_sql_references(self, sql: str, current_layer: str) -> str:
        """Process table references in SQL, converting them to schema.table format"""
        processed_sql = sql
        
                
        raw_table_pattern = r'\braw\.(\w+)'
        processed_sql = re.sub(raw_table_pattern, r'raw.\1', processed_sql, flags=re.IGNORECASE)
        
                
        staging_pattern = r'\b(?<!\.)\bstg_(\w+)__(\w+)'
        processed_sql = re.sub(staging_pattern, r'staging.stg_\1__\2', processed_sql, flags=re.IGNORECASE)
        
                
        intermediate_pattern = r'\b(?<!\.)\bint_(\w+)__(\w+)'
        processed_sql = re.sub(intermediate_pattern, r'intermediate.int_\1__\2', processed_sql, flags=re.IGNORECASE)
        
                
        marts_pattern = r'\b(?<!staging\.)(?<!intermediate\.)ad_reporting__(\w+)'
        processed_sql = re.sub(marts_pattern, r'marts.ad_reporting__\1', processed_sql, flags=re.IGNORECASE)
        
        return processed_sql
    
    def _execute_layer(self, layer_name: str) -> bool:
        """Execute all tables for a specific layer"""
        layer_config = self.config['layers'][layer_name]
        tables = layer_config.get('tables', [])
        
                
        execution_order = self.config.get('execution_order', {}).get(layer_name, tables)
        
        logger.info(f"\n=== Executing {layer_name.upper()} Layer ===")
        logger.info(f"Description: {layer_config.get('description', '')}")
        
        success_count = 0
        total_count = len(execution_order)
        
        for table_name in execution_order:
            
            sql_file_path = self.sql_dir / layer_name / f"{table_name}.sql"
            
            if not sql_file_path.exists():
                logger.warning(f"Skipping {table_name}: SQL file not found ({sql_file_path})")
                continue
            
            sql_content = self._load_sql_file(sql_file_path)
            if self._execute_sql(sql_content, table_name, layer_name):
                success_count += 1
        
        logger.info(f"{layer_name.upper()} layer complete: {success_count}/{total_count} tables created successfully")
        return success_count == total_count

    def run_pipeline(self, layers: Optional[List[str]] = None) -> bool:
        """Run the complete data pipeline"""
        if layers is None:
            layers = list(self.config['layers'].keys())
        
        logger.info("🚀 Starting Ad Data Pipeline execution")
        logger.info(f"Source database: {self.source_db_path}")
        logger.info(f"Target database: {self.target_db_path}")
        
                
        if not os.path.exists(self.target_db_path):
            logger.info("Target database does not exist, starting initialization...")
            try:
                
                self._initialize_database()
            except Exception as e:
                logger.error(f"Failed to initialize database: {e}")
                logger.info("Attempting to continue with existing database...")
        else:
            logger.info("Found existing target database, using it directly...")
            
            self.conn = duckdb.connect(self.target_db_path)
        
        self._clean_generated_tables()
        
                
        self._create_schemas()
        
        total_success = True
        
        for layer_name in layers:
            if layer_name not in self.config['layers']:
                logger.error(f"Unknown layer: {layer_name}")
                total_success = False
                continue
            
            dependencies = self.config['layers'][layer_name].get('dependencies', [])
            for dep in dependencies:
                if dep not in layers[:layers.index(layer_name)]:
                    logger.warning(f"{layer_name} depends on {dep}, but {dep} has not been executed yet")
            
            layer_success = self._execute_layer(layer_name)
            total_success = total_success and layer_success
        
        if total_success:
            logger.info("\n🎉 Ad Data Pipeline executed successfully!")
            logger.info(f"✨ The complete data warehouse has been generated: {self.target_db_path}")
            self._show_final_stats()
        else:
            logger.error("\n❌ Errors occurred during pipeline execution")
        
        if self.conn:
            self.conn.close()
        
        return total_success
    
    def _show_final_stats(self):
        """Display final statistics - shown by schema layer"""
        logger.info("\n=== Final Data Statistics (by Schema Layer) ===")
        
                
        for layer_name, schema_name in self.schema_config.items():
            try:
                
                tables_result = self.conn.execute(f"""SELECT table_name 
                                                     FROM information_schema.tables 
                                                     WHERE table_schema = '{schema_name}'""").fetchall()
                schema_tables = [table[0] for table in tables_result]
                
                if schema_tables:
                    logger.info(f"\n{layer_name.upper()} Layer (Schema: {schema_name}):")
                    for table in sorted(schema_tables):
                        try:
                            count = self.conn.execute(f"SELECT COUNT(*) FROM {schema_name}.{table}").fetchone()[0]
                            logger.info(f"  ✓ {schema_name}.{table}: {count:,} rows")
                        except Exception as e:
                            logger.info(f"  ✗ {schema_name}.{table}: Query failed - {e}")
                else:
                    logger.info(f"\n{layer_name.upper()} Layer (Schema: {schema_name}): No tables")
                    
            except Exception as e:
                logger.error(f"Failed to query schema {schema_name}: {e}")

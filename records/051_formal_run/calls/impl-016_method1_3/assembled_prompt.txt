WITH employee_base AS (
    SELECT worker_id, source_relation, hire_date, departure_date, is_employed,
           days_as_worker, COALESCE(departure_date, CURRENT_DATE)::DATE AS analysis_end_date
    FROM intermediate.int_workday__worker_details
)
SELECT worker_id, source_relation, hire_date, departure_date, is_employed, days_as_worker,
       analysis_end_date,
       CASE WHEN days_as_worker >= 30 OR is_employed THEN 1 ELSE 0 END::INTEGER AS retained_30_days,
       CASE WHEN days_as_worker >= 90 OR is_employed THEN 1 ELSE 0 END::INTEGER AS retained_90_days,
       CASE WHEN days_as_worker >= 365 OR is_employed THEN 1 ELSE 0 END::INTEGER AS retained_1_year,
       CASE WHEN days_as_worker >= 1095 OR is_employed THEN 1 ELSE 0 END::INTEGER AS retained_3_years,
       CASE WHEN is_employed = FALSE THEN 'Churned' WHEN days_as_worker < 90 THEN 'High Risk'
            WHEN days_as_worker < 365 THEN 'Medium Risk' WHEN days_as_worker < 730 THEN 'Low Risk' ELSE 'Stable' END AS churn_risk_category,
       CASE WHEN days_as_worker < 90 THEN 'New Hire (0-3 months)' WHEN days_as_worker < 365 THEN 'Junior (3-12 months)'
            WHEN days_as_worker < 730 THEN 'Experienced (1-2 years)' WHEN days_as_worker < 1825 THEN 'Senior (2-5 years)' ELSE 'Veteran (5+ years)' END AS employee_maturity_segment,
       CASE WHEN is_employed = FALSE THEN CASE WHEN days_as_worker < 90 THEN 'Early Termination'
            WHEN days_as_worker < 365 THEN 'Short-term Departure' WHEN days_as_worker < 730 THEN 'Mid-term Departure' ELSE 'Long-term Departure' END
            ELSE 'Active Employee' END AS departure_category,
       CASE WHEN days_as_worker > 0 THEN ROUND(days_as_worker / 365.0, 2) ELSE 0 END::DOUBLE AS tenure_years,
       CASE WHEN hire_date IS NULL THEN NULL ELSE strftime(hire_date, '%Y') || '-Q' || CAST(((month(hire_date)-1)//3)+1 AS VARCHAR) END AS hire_quarter_cohort,
       year(hire_date)::BIGINT AS hire_year_cohort,
       CASE WHEN hire_date IS NULL OR (is_employed IS NULL AND departure_date IS NULL) THEN NULL
            WHEN is_employed THEN year(CURRENT_DATE)-year(hire_date) ELSE year(departure_date)-year(hire_date) END::BIGINT AS calendar_years_worked,
       CASE WHEN is_employed = FALSE THEN 0 WHEN days_as_worker < 30 THEN 10 WHEN days_as_worker < 90 THEN 25
            WHEN days_as_worker < 365 THEN 50 WHEN days_as_worker < 730 THEN 75 WHEN days_as_worker < 1825 THEN 90 ELSE 100 END::INTEGER AS retention_stability_score,
       month(hire_date)::BIGINT AS hire_month,
       CASE WHEN hire_date IS NULL THEN NULL ELSE (dayofweek(hire_date))::BIGINT END AS hire_day_of_week,
       CASE WHEN hire_date IS NULL THEN NULL WHEN month(hire_date) IN (12,1,2) THEN 'Winter'
            WHEN month(hire_date) IN (3,4,5) THEN 'Spring' WHEN month(hire_date) IN (6,7,8) THEN 'Summer' ELSE 'Fall' END AS hire_season,
       CASE WHEN hire_date > CURRENT_DATE THEN 'ERROR: Future hire date'
            WHEN departure_date < hire_date THEN 'ERROR: Departure before hire'
            WHEN days_as_worker < 0 THEN 'ERROR: Negative tenure' ELSE 'VALID' END AS data_quality_flag
FROM employee_base

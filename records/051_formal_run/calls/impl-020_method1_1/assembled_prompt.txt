#!/usr/bin/env python3
import os
import yaml
import duckdb
from pathlib import Path
import logging
from typing import List, Optional

logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')
logger = logging.getLogger(__name__)

class DataPipeline:
    def __init__(self,
                 source_db_path: str = "intercom_start.duckdb",
                 target_db_path: str = "intercom.duckdb",
                 config_path: str = "config/layer_dependencies.yaml"):
        self.source_db_path = source_db_path
        self.target_db_path = target_db_path
        self.config_path = config_path
        self.sql_dir = Path("sql")
        self.conn: Optional[duckdb.DuckDBPyConnection] = None
        self.config = self._load_config()
        self.schemas = {
            'raw': 'raw',
            'staging': 'staging',
            'intermediate': 'intermediate',
            'marts': 'marts',
        }

    def _load_config(self) -> dict:
        with open(self.config_path, 'r', encoding='utf-8') as f:
            return yaml.safe_load(f)

    def _create_schemas(self):
        for schema in self.schemas.values():
            self.conn.execute(f"CREATE SCHEMA IF NOT EXISTS {schema}")
            logger.info(f"schema ready: {schema}")

    def _clean_non_raw(self):
        for schema in ['staging', 'intermediate', 'marts']:
            try:
                self.conn.execute(f"DROP SCHEMA IF EXISTS {schema} CASCADE")
                logger.info(f"dropped schema: {schema}")
            except Exception as e:
                logger.warning(f"drop {schema} failed: {e}")

    def _import_raw(self):
        logger.info("importing raw from source DB")
        src = duckdb.connect(self.source_db_path)
        try:
            tables = [r[0] for r in src.execute("SHOW TABLES").fetchall()]
            if len(tables) == 0:
                raise RuntimeError(f"No tables found in source DB {self.source_db_path}; CSV import is disabled")
            logger.info(f"found {len(tables)} raw tables")
            for t in tables:
                cols = src.execute(f"DESCRIBE {t}").fetchall()
                col_defs = []
                for name, typ, *_ in cols:
                    safe = name if name.lower() not in {'end','start','order','group','select','from','where','type'} else f'"{name}"'
                    col_defs.append(f"{safe} {typ}")
                self.conn.execute(f"CREATE OR REPLACE TABLE raw.{t} ({', '.join(col_defs)})")
                data = src.execute(f"SELECT * FROM {t}").fetchall()
                if data:
                    placeholders = ', '.join(['?' for _ in cols])
                    self.conn.executemany(f"INSERT INTO raw.{t} VALUES ({placeholders})", data)
                logger.info(f"raw.{t} rows: {len(data)}")
        finally:
            src.close()

    def _load_sql(self, path: Path) -> str:
        with open(path, 'r', encoding='utf-8') as f:
            return f.read()

    def _execute_table(self, layer: str, table: str) -> bool:
        sql_path = self._resolve_sql_path(layer, table)
        if not sql_path:
            logger.error(f"SQL not found for {layer}.{table}")
            return False
        sql = self._load_sql(sql_path)
        full = f"{self.schemas[layer]}.{table}"
        try:
            self.conn.execute(f"CREATE OR REPLACE TABLE {full} AS ({sql})")
            cnt = self.conn.execute(f"SELECT COUNT(*) FROM {full}").fetchone()[0]
            logger.info(f"built {full}: {cnt} rows")
            return True
        except Exception as e:
            logger.error(f"build {full} failed: {e}")
            return False

    def _resolve_sql_path(self, layer: str, table: str) -> Optional[Path]:
        p = self.sql_dir / layer / f"{table}.sql"
        return p if p.exists() else None

    def run(self, layers: Optional[List[str]] = None) -> bool:
        if layers is None:
            layers = list(self.config['layers'].keys())
        created = os.path.exists(self.target_db_path)
        self.conn = duckdb.connect(self.target_db_path)
        if created:
            logger.info("existing target DB found")
        self._clean_non_raw()
        self._create_schemas()
        self._import_raw()
        ok = True
        for layer in layers:
            logger.info(f"=== running layer: {layer} ===")
            deps = self.config['layers'][layer].get('dependencies', [])
            order = self.config.get('execution_order', {}).get(layer, self.config['layers'][layer].get('tables', []))
            allowed = set(self.config['layers'][layer].get('tables', []))
            ordered_tables = [t for t in order if t in allowed]
            for tbl in ordered_tables:
                if not self._execute_table(layer, tbl):
                    ok = False
        return ok

    pipe = DataPipeline()

#!/usr/bin/env python3
"""
Utility to execute a single SQL model against lever_start.duckdb (or a custom db).

This helps iterate on one SQL file without running the whole config-driven pipeline:

    python test_single_sql.py --table stg_lever__application --layer staging
"""

from __future__ import annotations

import argparse
import sys
from pathlib import Path
from typing import Optional, Tuple

import duckdb

SQL_ROOT = Path("sql")
INTERMEDIATE_SUBDIRS = ["opportunity", "contact", "interview", "posting", "requisition"]
LAYER_SCHEMAS = {
    "raw": "raw",
    "staging": "staging",
    "intermediate": "intermediate",
    "marts": "marts",
}
DB_PATH = Path("intercom_start.duckdb")


def _candidate_paths(table: str, layer: Optional[str]) -> Tuple[Tuple[Path, str], ...]:
    """
    Build all possible SQL paths for a table, optionally restricting to a single layer.
    Returns tuples of (path, layer_name) in priority order.
    """
    search_layers = [layer] if layer else list(LAYER_SCHEMAS.keys())
    candidates = []

    for layer_name in search_layers:
        base = SQL_ROOT / layer_name / f"{table}.sql"
        candidates.append((base, layer_name))

        if layer_name == "intermediate":
            for subdir in INTERMEDIATE_SUBDIRS:
                candidates.append(
                    (SQL_ROOT / layer_name / subdir / f"{table}.sql", layer_name)
                )

    return tuple(candidates)

def locate_sql(table: str, layer: Optional[str]) -> Tuple[Path, str]:
    """Locate the SQL file for `table` and return (path, layer)."""
    for candidate, candidate_layer in _candidate_paths(table, layer):
        if candidate.exists():
            return candidate, candidate_layer
    searched = "\n  ".join(str(path) for path, _ in _candidate_paths(table, layer))
    raise FileNotFoundError(
        f"SQL file for {table} not found. Checked paths:\n  {searched}"
    )

def ensure_schema(conn: duckdb.DuckDBPyConnection, schema: str) -> None:
    conn.execute(f"CREATE SCHEMA IF NOT EXISTS {schema}")


def execute_sql(
    conn: duckdb.DuckDBPyConnection,
    sql_text: str,
    schema: str,
    table: str,
) -> int:
    full_table_name = f"{schema}.{table}"
    conn.execute(f"DROP TABLE IF EXISTS {full_table_name}")
    conn.execute(f"CREATE TABLE {full_table_name} AS ({sql_text})")
    row_count = conn.execute(f"SELECT COUNT(*) FROM {full_table_name}").fetchone()[0]
    return row_count


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(
        description=("Execute a single SQL model against lever_start.duckdb to validate logic without running the full pipeline.")
    )
    parser.add_argument("--table", required=True, help="Table/model name without .sql, e.g. stg_lever__application")
    parser.add_argument("--layer", choices=LAYER_SCHEMAS.keys(), help="Layer where the table lives; auto-detected when omitted")
    return parser.parse_args()

def main() -> int:
    args = parse_args()
    try:
        sql_path, layer = locate_sql(args.table, args.layer)
    except FileNotFoundError as exc:
        print(exc, file=sys.stderr)
        return 1

    schema = LAYER_SCHEMAS[layer]
    if not DB_PATH.exists():
        print(f"DuckDB file not found: {DB_PATH}", file=sys.stderr)
        return 1

    sql_text = sql_path.read_text(encoding="utf-8")

    print(f"Connecting to database: {DB_PATH}")
    conn = duckdb.connect(str(DB_PATH))
    ensure_schema(conn, schema)

    print(f"Executing SQL: {sql_path} -> {schema}.{args.table}")
    row_count = execute_sql(conn, sql_text, schema, args.table)
    print(f"Done: {schema}.{args.table} contains {row_count} rows")

    conn.close()
    return 0

if __name__ == "__main__":
    raise SystemExit(main())

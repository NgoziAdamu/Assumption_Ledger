        logger.info(f"✓ Layered database initialization complete: {self.target_db_path}")
    
    def _create_schemas(self):
        """Create the schemas for the data layers"""
        logger.info("🏗️ Creating Schema layered architecture")
        
        for layer, schema_name in self.schema_config.items():
            try:
                self.conn.execute(f"CREATE SCHEMA IF NOT EXISTS {schema_name}")
                logger.info(f"  ✓ Created schema: {schema_name}")
            except Exception as e:
                logger.error(f"  ✗ Failed to create schema {schema_name}: {e}")
                raise
    
    def _import_raw_data(self):
        """Import raw data from the source database to the raw schema"""
        logger.info("📥 Importing raw data into the raw schema")
        
                
        try:
            source_conn = duckdb.connect(self.source_db_path)
        except Exception as e:
            logger.error(f"Unable to connect to source database {self.source_db_path}: {e}")
            raise
        
        try:
            
            tables_result = source_conn.execute("SELECT table_name FROM information_schema.tables WHERE table_schema = 'raw'").fetchall()
            source_tables = [table[0] for table in tables_result]
            
            logger.info(f"Found {len(source_tables)} raw tables")
            
                        
            for table_name in source_tables:
                try:
                    
                    data = source_conn.execute(f"SELECT * FROM raw.{table_name}").fetchall()
                    columns_info = source_conn.execute(f"DESCRIBE raw.{table_name}").fetchall()
                    
                                        
                    column_defs = []
                    for col_info in columns_info:
                        col_name, col_type = col_info[0], col_info[1]
                        if col_name.lower() in ['end', 'start', 'order', 'group', 'select', 'from', 'where']:
                            safe_col_name = f'"{col_name}"'
                        else:
                            safe_col_name = col_name
                        column_defs.append(f"{safe_col_name} {col_type}")
                    
                    create_sql = f"CREATE OR REPLACE TABLE raw.{table_name} ({', '.join(column_defs)})"
                    self.conn.execute(create_sql)
                    
                                        
                    if data:
                        placeholders = ', '.join(['?' for _ in columns_info])
                        insert_sql = f"INSERT INTO raw.{table_name} VALUES ({placeholders})"
                        self.conn.executemany(insert_sql, data)
                    
                    logger.info(f"  ✓ raw.{table_name}: {len(data)} rows")
                    
                except Exception as e:
                    logger.error(f"  ✗ Failed to import table {table_name}: {e}")
        
        finally:
            source_conn.close()
        
        logger.info("✓ Raw data import complete")
    
    def _load_sql_file(self, file_path: Path) -> str:
        """Load the content of a SQL file"""
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                return f.read()
        except FileNotFoundError:
            logger.error(f"SQL file {file_path} not found")
            raise
    
    def _execute_sql(self, sql: str, table_name: str, layer_name: str) -> bool:
        """Execute SQL and create a table in the specified schema"""
        try:
            
            target_schema = self.schema_config[layer_name]
            full_table_name = f"{target_schema}.{table_name}"
            
                        
            processed_sql = self._process_sql_references(sql, layer_name)
            
                        
            create_table_sql = f"CREATE OR REPLACE TABLE {full_table_name} AS ({processed_sql})"
            
            logger.info(f"Executing: {full_table_name}")
            self.conn.execute(create_table_sql)
            
                        
            count_result = self.conn.execute(f"SELECT COUNT(*) FROM {full_table_name}").fetchone()
            row_count = count_result[0] if count_result else 0
            
            logger.info(f"✓ {full_table_name}: {row_count:,} rows")
            return True
            
        except Exception as e:
            logger.error(f"✗ {full_table_name}: {e}")
            return False
    
    def _process_sql_references(self, sql: str, current_layer: str) -> str:
        """Process table references in SQL, converting them to schema.table format"""
        import re
        processed_sql = sql
        
                
        cte_pattern = r'(\w+)\s+as\s*\('
        cte_names = set(re.findall(cte_pattern, sql, re.IGNORECASE))
        
                
        raw_table_mappings = {
            'account_history_data': 'raw.account_history_data',
            'account_stats_data': 'raw.account_stats_data',
            'campaign_history_data': 'raw.campaign_history_data',
            'campaign_stats_data': 'raw.campaign_stats_data',
            'ad_group_history_data': 'raw.ad_group_history_data',
            'ad_group_stats_data': 'raw.ad_group_stats_data',
            'ad_history_data': 'raw.ad_history_data',
            'ad_stats_data': 'raw.ad_stats_data',
            'ad_group_criterion_history_data': 'raw.ad_group_criterion_history_data',
            'keyword_stats_data': 'raw.keyword_stats_data',
            'search_term_keyword_stats_data': 'raw.search_term_keyword_stats_data'
        }
        
                
        for table_name, schema_table in raw_table_mappings.items():
            if table_name not in cte_names:
                pattern = r'\bfrom\s+' + re.escape(table_name) + r'\b'
                replacement = f'from {schema_table}'
                processed_sql = re.sub(pattern, replacement, processed_sql, flags=re.IGNORECASE)
        
        processed_sql = re.sub(r'\bfrom\s+stg_google_ads__', 'from staging.stg_google_ads__', processed_sql, flags=re.IGNORECASE)
        
                
        processed_sql = re.sub(r'\bfrom\s+int_google_ads__', 'from intermediate.int_google_ads__', processed_sql, flags=re.IGNORECASE)
        
                
        processed_sql = re.sub(r'\bfrom\s+google_ads__', 'from marts.google_ads__', processed_sql, flags=re.IGNORECASE)
        
        return processed_sql
    
    def _execute_layer(self, layer_name: str) -> bool:
        """Execute all tables for a specific layer"""
        layer_config = self.config['layers'][layer_name]
        tables = layer_config.get('tables', [])
        
                
        execution_order = self.config.get('execution_order', {}).get(layer_name, tables)
        
        logger.info(f"\n=== Executing {layer_name.upper()} Layer ===")
        logger.info(f"Description: {layer_config.get('description', '')}")
        
        success_count = 0
        total_count = len(execution_order)
        
        for table_name in execution_order:
            
            sql_file_path = self.sql_dir / layer_name / f"{table_name}.sql"
            
            if not sql_file_path.exists():
                logger.warning(f"Skipping {table_name}: SQL file does not exist ({sql_file_path})")
                continue
            
            sql_content = self._load_sql_file(sql_file_path)
            if self._execute_sql(sql_content, table_name, layer_name):
                success_count += 1
        
        logger.info(f"{layer_name.upper()} Layer complete: {success_count}/{total_count} tables created successfully")

rolling_metrics AS (
    SELECT
        *,
        avg(spend) OVER (PARTITION BY account_id, source_relation ORDER BY date_day ROWS BETWEEN 6 PRECEDING AND CURRENT ROW) AS spend_7d_avg,
        avg(clicks) OVER (PARTITION BY account_id, source_relation ORDER BY date_day ROWS BETWEEN 6 PRECEDING AND CURRENT ROW) AS clicks_7d_avg,
        avg(impressions) OVER (PARTITION BY account_id, source_relation ORDER BY date_day ROWS BETWEEN 6 PRECEDING AND CURRENT ROW) AS impressions_7d_avg,
        lag(spend, 30) OVER (PARTITION BY account_id, source_relation ORDER BY date_day) AS spend_lag_30d,
        lag(conversions, 30) OVER (PARTITION BY account_id, source_relation ORDER BY date_day) AS conversions_lag_30d,
        min(date_day) OVER (PARTITION BY account_id) AS first_activity_date
    FROM daily_stats
),
metrics AS (
    SELECT *,
        round(CASE WHEN impressions > 0 THEN clicks::DOUBLE / impressions * 100 ELSE 0 END, 4)::FLOAT AS ctr,
        round(CASE WHEN clicks > 0 THEN spend / clicks ELSE 0 END, 4) AS cpc,
        round(CASE WHEN impressions > 0 THEN spend / impressions * 1000 ELSE 0 END, 4) AS cpm,
        round(CASE WHEN conversions > 0 THEN spend / conversions END, 4) AS cost_per_conversion,
        round(CASE WHEN spend > 0 AND conversions_value > 0 THEN conversions_value / spend ELSE 0 END, 4) AS roas,
        greatest(date_diff('day', first_activity_date, date_day), 0) AS days_since_first_activity
    FROM rolling_metrics
),
scored AS (
    SELECT *,
        round(CASE WHEN spend_lag_30d > 0 THEN (spend - spend_lag_30d) / spend_lag_30d * 100 END, 2) AS spend_mom_change_pct,
        round(CASE WHEN conversions_lag_30d > 0 THEN (conversions - conversions_lag_30d) / conversions_lag_30d * 100 END, 2) AS conversions_mom_change_pct,
        CASE WHEN ctr IS NULL OR roas IS NULL OR clicks_7d_avg IS NULL THEN NULL ELSE least(100.0, greatest(0.0, ctr * 10 + least(roas * 10, 50) + CASE WHEN cost_per_conversion < 50 THEN 20 ELSE 0 END + CASE WHEN clicks_7d_avg > 100 THEN 10 ELSE clicks_7d_avg / 10 END)) END AS performance_score,
        CASE WHEN days_since_first_activity < 30 THEN 'New Account' WHEN days_since_first_activity < 90 THEN 'Growing Account' WHEN days_since_first_activity < 365 THEN 'Established Account' ELSE 'Mature Account' END AS account_maturity_stage,
        (spend > 0 AND clicks = 0) AS has_spend_no_clicks_flag,
        (clicks > impressions) AS invalid_metrics_flag,
        (spend > 10000) AS high_spend_alert_flag
    FROM metrics
),
with_tier AS (
    SELECT *, CASE WHEN performance_score >= 80 THEN 'Excellent' WHEN performance_score >= 60 THEN 'Good' WHEN performance_score >= 40 THEN 'Average' WHEN performance_score >= 20 THEN 'Below Average' ELSE 'Poor' END AS performance_tier
    FROM scored
)
